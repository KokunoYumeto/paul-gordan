\GordanTopFolio{7}
must necessarily be expressed as functions of the angle $\varphi$. The quantity $l$ is determined from the equation
\[
 \sin l=\sin l_0\cos\varphi,
\]
or from these:
\[
\begin{aligned}
\cos l&=\sqrt{1-\sin^2l_0\cos^2\varphi}=\cos l_0\sqrt{1+\tg^2l_0\sin^2\varphi},\\
\tg l&={\tg l_0\cos\varphi\over\sqrt{1+\tg^2l_0\sin^2\varphi}}
={\cos\varphi\over\sqrt{\cot^2l_0+\sin^2\varphi}},\\
\sqrt{\sin^2l_0-\sin^2l}&=\sin l_0\sin\varphi,\\
\sqrt{1-e^2\cos^2l}&=\sqrt{1-e^2+e^2\sin^2l_0\cos^2\varphi}
=\sqrt{1-e^2\cos^2l_0-e^2\sin^2l_0\sin^2\varphi}.
\end{aligned}
\]
Substituting these values into the equations proposed in Chapter I, \S\ 7, and Chapter II, \S\ 4 and \S\ 6, we obtain
\[
 \sqrt{1-e^2}\,\tg B={\cos\varphi\over\sqrt{\cot^2l_0+\sin^2\varphi}},
\]
\[
 \sin B={\sin l_0\cos\varphi\over\sqrt{1-e^2+e^2\sin^2l_0\cos^2\varphi}},
\]
\[
 \cos B=\cos l_0\sqrt{1-e^2}\sqrt{\frac{1+\tg^2l_0\sin^2\varphi}{1-e^2+e^2\sin^2l_0\cos^2\varphi}},
\]
\[
 \tg T={\cot l_0\over\sin\varphi},\qquad
 \sin T={1\over\sqrt{1+\tg^2l_0\sin^2\varphi}},\qquad
 \cos T={\sin\varphi\over\sqrt{\cot^2l_0+\sin^2\varphi}},
\]
\[
 s=a\int_0^\varphi d\varphi\sqrt{1-e^2+e^2\sin^2l_0\cos^2\varphi}
 =a\int_0^\varphi d\varphi\sqrt{1-e^2\cos^2l_0-e^2\sin^2l_0\sin^2\varphi},
\]
\[
 \psi=\pm\int_0^\varphi d\varphi\,{\cos l_0\over\sin^2l_0}\,
 {\sqrt{1-e^2\cos^2l_0-e^2\sin^2l_0\sin^2\varphi}\over\cot^2l_0+\sin^2\varphi}.
\]
The last two equations, putting
\[
 {e\sin l_0\over\sqrt{1-e^2\cos^2l_0}}=e\sin B_0=x,
\]
and
\[
 \sqrt{1-e^2\sin^2B_0}={\cos B_0\over\cos l_0}=x',
\]
are transformed into
\[
 s={b\over x'}\int_0^\varphi d\varphi\sqrt{1-x^2\sin^2\varphi},
\]
\[
 \psi=\pm{\cot l_0\over\sin B_0}\int_0^\varphi d\varphi\,{\sqrt{1-x^2\sin^2\varphi}\over\cot^2l_0+\sin^2\varphi}.
\]
